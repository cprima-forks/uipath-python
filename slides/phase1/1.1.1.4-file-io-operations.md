## Opening and Closing Files

```python
# ❌ BAD: Manual closing (error-prone)
file = open('data.txt', 'r')
content = file.read()
file.close()  # What if an error occurs before this?

# ✅ GOOD: Context manager (automatic cleanup)
with open('data.txt', 'r') as file:
    content = file.read()
# File automatically closed here
```

**Always use `with` statements for files!**

---

## File Modes

```python
# Reading
'r'   # Read (default) - Error if file doesn't exist
'rb'  # Read binary

# Writing
'w'   # Write - OVERWRITES existing file!
'wb'  # Write binary
'x'   # Exclusive creation - Error if file exists

# Appending
'a'   # Append - Adds to end of file
'ab'  # Append binary

# Read and Write
'r+'  # Read and write
'w+'  # Write and read (overwrites)
```

## Reading Text Files

```python
# Read entire file as string
with open('document.txt', 'r') as f:
    content = f.read()
print(content)

# Read line by line (memory efficient)
with open('document.txt', 'r') as f:
    for line in f:
        print(line.strip())

# Read all lines into a list
with open('document.txt', 'r') as f:
    lines = f.readlines()
# ['line 1\n', 'line 2\n', 'line 3\n']
```

---

## Writing Text Files

```python
# Write (overwrites entire file!)
with open('output.txt', 'w') as f:
    f.write('Hello, World!\n')
    f.write('Second line\n')

# Append (adds to end)
with open('log.txt', 'a') as f:
    f.write('New log entry\n')

# Write multiple lines
lines = ['Line 1\n', 'Line 2\n', 'Line 3\n']
with open('output.txt', 'w') as f:
    f.writelines(lines)
```

**⚠️ Mode 'w' erases existing content!**

## Working with CSV Files

with open('data.csv', 'r') as f:

## Real-World Example: Processing Reports

```python
import csv

def process_sales_report(input_file, output_file):
    """Extract high-value sales from report."""
    high_value_sales = []

    with open(input_file, 'r') as f:
        reader = csv.DictReader(f)
        for row in reader:
            if float(row['amount']) > 1000:
                high_value_sales.append(row)

    # Write filtered results
    with open(output_file, 'w', newline='') as f:
        if high_value_sales:
            fieldnames = high_value_sales[0].keys()
            writer = csv.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            writer.writerows(high_value_sales)

    return len(high_value_sales)
```

## JSON: dumps vs dump

```python
import json

data = {'name': 'Alice', 'age': 30}

# dumps() - returns JSON STRING
json_string = json.dumps(data)
print(json_string)  # '{"name": "Alice", "age": 30}'
print(type(json_string))  # <class 'str'>

# dump() - writes to FILE
with open('data.json', 'w') as f:
    json.dump(data, f)

# loads() - parses JSON STRING
parsed = json.loads(json_string)

# load() - reads from FILE
with open('data.json', 'r') as f:
    loaded = json.load(f)
```

**Remember: 's' suffix = string, no 's' = file**

## Real-World Example: Agent Configuration

```python
import json
from pathlib import Path

class AgentConfig:
    """Manage agent configuration."""

    def __init__(self, config_path='config.json'):
        self.config_path = Path(config_path)
        self.config = self.load_config()

    def load_config(self):
        """Load configuration from JSON file."""
        if self.config_path.exists():
            with open(self.config_path, 'r') as f:
                return json.load(f)
        return self.get_defaults()

    def save_config(self):
        """Save configuration to JSON file."""
        with open(self.config_path, 'w') as f:
            json.dump(self.config, f, indent=2)

    def get_defaults(self):
        return {
            'timeout': 30,
            'max_retries': 3,
            'confidence_threshold': 0.85
        }
```

## Cross-Platform Path Handling

```python
import os

# ❌ BAD: Platform-specific
path = 'data\\files\\document.txt'  # Windows only!

# ✅ BETTER: os.path.join (works everywhere)
path = os.path.join('data', 'files', 'document.txt')

# ✅ BEST: pathlib (modern, object-oriented)
from pathlib import Path
path = Path('data') / 'files' / 'document.txt'
```

**pathlib is the modern way**

## Real-World Example: Document Processing

```python
from pathlib import Path
import json

def process_documents(input_dir, output_dir):
    """Process all JSON documents in directory."""
    input_path = Path(input_dir)
    output_path = Path(output_dir)

    # Ensure output directory exists
    output_path.mkdir(parents=True, exist_ok=True)

    results = []

    # Process each JSON file
    for json_file in input_path.glob('*.json'):
        try:
            # Read document
            with open(json_file, 'r') as f:
                data = json.load(f)

            # Process (example: extract key info)
            processed = {
                'filename': json_file.name,
                'id': data.get('id'),
                'status': 'processed'
            }
            results.append(processed)

        except Exception as e:
            print(f"Error processing {json_file}: {e}")

    # Save results
    with open(output_path / 'results.json', 'w') as f:
        json.dump(results, f, indent=2)

    return len(results)
```

## File Error Handling

```python
from pathlib import Path

def safe_read_file(file_path):
    """Safely read a file with error handling."""
    try:
        with open(file_path, 'r') as f:
            return f.read()

    except FileNotFoundError:
        print(f"File not found: {file_path}")
        return None

    except PermissionError:
        print(f"Permission denied: {file_path}")
        return None

    except UnicodeDecodeError:
        print(f"Encoding error (try binary mode): {file_path}")
        return None

    except Exception as e:
        print(f"Unexpected error: {e}")
        return None
```

## EAFP vs LBYL

```python
# LBYL: Look Before You Leap
if os.path.exists(file_path):
    with open(file_path) as f:
        data = f.read()
# ⚠️ Race condition: file could be deleted between check and open

# EAFP: Easier to Ask Forgiveness than Permission (Pythonic!)
try:
    with open(file_path) as f:
        data = f.read()
except FileNotFoundError:
    data = None
# ✅ No race condition, handles all errors consistently
```

**Python philosophy: EAFP over LBYL**

## Binary Files

```python
# Reading binary files (images, PDFs, etc.)
with open('image.png', 'rb') as f:
    data = f.read()  # bytes object

# Writing binary files
with open('copy.png', 'wb') as f:
    f.write(data)

# Why binary mode?
# - Text mode does encoding/decoding (UTF-8, etc.)
# - Binary mode reads/writes raw bytes
# - Use binary for: images, videos, PDFs, executables
```

## Real-World Example: File Copying

```python
from pathlib import Path

def copy_file(source, destination):
    """Copy a file safely."""
    source_path = Path(source)
    dest_path = Path(destination)

    # Check source exists
    if not source_path.exists():
        raise FileNotFoundError(f"Source not found: {source}")

    # Ensure destination directory exists
    dest_path.parent.mkdir(parents=True, exist_ok=True)

    # Copy file (binary mode for all file types)
    with open(source_path, 'rb') as src:
        with open(dest_path, 'wb') as dst:
            dst.write(src.read())

    return dest_path

# Usage
copy_file('data/input.pdf', 'backup/input.pdf')
```

## Character Encoding

```python
# Default encoding (UTF-8)
with open('file.txt', 'r') as f:
    content = f.read()

# Specify encoding
with open('file.txt', 'r', encoding='utf-8') as f:
    content = f.read()

# Handle encoding errors
with open('file.txt', 'r', encoding='utf-8', errors='ignore') as f:
    content = f.read()

# Common encodings:
# - utf-8 (universal, recommended)
# - ascii (English only)
# - latin-1 (Western European)
# - utf-16 (Windows sometimes uses this)
```

## File Operations Best Practices

**DO:**
- ✅ Use `with` statements (context managers)
- ✅ Use pathlib for path operations
- ✅ Specify encoding explicitly
- ✅ Use binary mode for non-text files
- ✅ Handle errors gracefully
- ✅ Close files (with does this automatically)

**DON'T:**
- ❌ Forget to close files manually
- ❌ Use mode 'w' without backing up data
- ❌ Hardcode path separators (/ or \)
- ❌ Ignore encoding issues
- ❌ Assume files exist without checking

---

## Common Pitfalls

```python
# ❌ Pitfall 1: Not closing files
f = open('file.txt')
data = f.read()
# File never closed! (use 'with' instead)

# ❌ Pitfall 2: Mode 'w' overwrites without warning
with open('important.txt', 'w') as f:
    f.write('Oops')  # All previous content lost!

# ❌ Pitfall 3: Platform-specific paths
path = 'data\\files\\doc.txt'  # Breaks on Linux/Mac

# ❌ Pitfall 4: Reading after read()
with open('file.txt') as f:
    content = f.read()
    lines = f.readlines()  # Returns [] (file pointer at end)
```

## File Pointer and seek()

```python
with open('data.txt', 'r') as f:
    # File pointer starts at position 0
    first_part = f.read(10)  # Read 10 bytes

    # File pointer now at position 10
    current_position = f.tell()  # 10

    # Reset to beginning
    f.seek(0)

    # Read again from start
    full_content = f.read()

    # Go to specific position
    f.seek(5)
    from_position_5 = f.read()
```

**Useful for large files or selective reading**

## Temporary Files

```python
import tempfile

# Temporary file (auto-deleted after use)
with tempfile.TemporaryFile(mode='w+') as f:
    f.write('Temporary data')
    f.seek(0)
    content = f.read()
# File deleted here

# Named temporary file (can open multiple times)
with tempfile.NamedTemporaryFile(mode='w', delete=False) as f:
    f.write('Data')
    temp_path = f.name

# Use temp file
with open(temp_path) as f:
    data = f.read()

# Clean up
os.unlink(temp_path)
```

## Real-World Example: Batch Processing

```python
from pathlib import Path
import json
import csv

def convert_json_to_csv(json_dir, output_csv):
    """Convert multiple JSON files to single CSV."""
    json_path = Path(json_dir)
    all_data = []

    # Read all JSON files
    for json_file in json_path.glob('*.json'):
        try:
            with open(json_file, 'r') as f:
                data = json.load(f)
                all_data.append(data)
        except Exception as e:
            print(f"Error: {json_file}: {e}")

    # Write to CSV
    if all_data:
        with open(output_csv, 'w', newline='') as f:
            fieldnames = all_data[0].keys()
            writer = csv.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            writer.writerows(all_data)

    return len(all_data)

# Usage
count = convert_json_to_csv('data/invoices', 'output/invoices.csv')
print(f"Converted {count} files")
```
