## Why Batch Execution?

### The Scale Problem

**Single Job Approach:**
- Process 10,000 transactions sequentially
- 2 seconds per transaction
- Total time: **5.5 hours**

**Batch Approach (10 parallel jobs):**
- 1,000 transactions per job
- All jobs run in parallel
- Total time: **~33 minutes**

**10x faster!**

## Batch Execution Defined

### What is Batch Execution?

**Starting multiple job instances concurrently to process work in parallel**

**Key Characteristics:**
- Multiple jobs run simultaneously
- Work is partitioned across jobs
- Results are aggregated after completion
- Failures are handled gracefully

## Data Partitioning

### Dividing Work

**Goal:** Non-overlapping, roughly equal partitions

```python
def partition_data(data, num_partitions):
    """Split data into equal-sized chunks"""
    chunk_size = len(data) // num_partitions

    partitions = []
    for i in range(num_partitions):
        start = i * chunk_size
        # Last partition gets remainder
        end = start + chunk_size if i < num_partitions - 1 else len(data)
        partitions.append(data[start:end])

    return partitions
```

```python
def start_batch_jobs_safe(orchestrator_url, token, folder_id,
                          release_key, input_batches):
    """Start batch jobs with error handling"""

    job_ids = []
```

```python
    """Determine optimal batch size"""

    batch_size = min(available_robots, max_batch_size)

    batch_size = min(batch_size, total_items)
```

## Concurrent Job Monitoring (Continued)

### Efficient Approach

**Poll all jobs in each iteration:**

```python
def wait_for_all_jobs(orchestrator_url, token, folder_id,
                       job_ids, poll_interval=3, timeout=600):
    """Monitor multiple jobs concurrently"""

    pending_jobs = set(job_ids)
    completed_jobs = {}
    start_time = time.time()

    while pending_jobs:
        # Check timeout
        if time.time() - start_time > timeout:
            raise TimeoutError(f"{len(pending_jobs)} jobs still pending")

        # Check each pending job
        for job_id in list(pending_jobs):
            job = get_job(orchestrator_url, token, folder_id, job_id)

            if job['State'] in ['Successful', 'Faulted', 'Stopped']:
                # Job finished
                completed_jobs[job_id] = job
                pending_jobs.remove(job_id)
                print(f"Job {job_id}: {job['State']}")

        # Sleep before next poll
        if pending_jobs:
            time.sleep(poll_interval)

    return completed_jobs
```

## Optimized Monitoring with OData

### Batch API Queries

**Instead of N API calls per poll, use 1:**

```python
def wait_for_all_jobs_optimized(orchestrator_url, token, folder_id,
                                 job_ids, poll_interval=5, timeout=600):
    """Monitor with single API call per poll"""

    pending_jobs = set(job_ids)
    completed_jobs = {}
    start_time = time.time()

    while pending_jobs:
        if time.time() - start_time > timeout:
            raise TimeoutError(f"{len(pending_jobs)} jobs still pending")

        # Build OData filter for all pending jobs
        id_list = "','".join(pending_jobs)
        filter_query = f"Key in ('{id_list}')"

        # Single API call for all jobs
        response = requests.get(
            f"{orchestrator_url}/odata/Jobs",
            headers={
                "Authorization": f"Bearer {token}",
                "X-UIPATH-OrganizationUnitId": str(folder_id)
            },
            params={"$filter": filter_query}
        )

        jobs = response.json()['value']

        # Check each job
        for job in jobs:
            if job['State'] in ['Successful', 'Faulted', 'Stopped']:
                completed_jobs[job['Key']] = job
                pending_jobs.remove(job['Key'])
                print(f"Job {job['Key']}: {job['State']}")

        if pending_jobs:
            time.sleep(poll_interval)

    return completed_jobs
```

## Result Aggregation

### Collecting Outputs

```python
def aggregate_batch_results(completed_jobs):
    """Aggregate results from all jobs"""

    aggregated = {
        'total_jobs': len(completed_jobs),
        'successful': 0,
        'faulted': 0,
        'stopped': 0,
        'outputs': [],
        'errors': []
    }

    for job_id, job in completed_jobs.items():
        state = job['State']

        if state == 'Successful':
            aggregated['successful'] += 1

            # Parse outputs
            outputs = get_job_outputs(job)
            if outputs:
                aggregated['outputs'].append({
                    'job_id': job_id,
                    'batch_id': outputs.get('batch_id'),
                    'data': outputs
                })

        elif state == 'Faulted':
            aggregated['faulted'] += 1

            # Parse error
            error = parse_error_info(job.get('Info', ''))
            aggregated['errors'].append({
                'job_id': job_id,
                'error': error
            })

        else:  # Stopped
            aggregated['stopped'] += 1

    return aggregated
```

## Result Aggregation (Continued)

### Combining Data

```python
def combine_batch_outputs(aggregated_results):
    """Combine outputs from all successful jobs"""

    all_processed_items = []
    total_count = 0

    for output in aggregated_results['outputs']:
        batch_data = output['data']

        # Extract results from this batch
        items = batch_data.get('processed_items', [])
        count = batch_data.get('processed_count', 0)

        all_processed_items.extend(items)
        total_count += count

    return {
        'combined_items': all_processed_items,
        'total_processed': total_count,
        'num_batches': len(aggregated_results['outputs'])
    }
```

## Error Handling in Batches

### Partial Failure Strategy

**Key Principle:** Don't let one failure stop all processing

**Approaches:**

1. **Continue on Error**
   - Complete all jobs even if some fail
   - Report partial success

2. **Fail Fast**
   - Stop starting new jobs if early failures occur
   - Use for critical dependencies

3. **Retry Failed Batches**
   - Identify failed batches
   - Retry with same or different inputs

## Error Handling in Batches (Continued)

```python
def analyze_batch_results(aggregated_results):
    """Analyze batch execution results"""

    total = aggregated_results['total_jobs']
    successful = aggregated_results['successful']
    faulted = aggregated_results['faulted']

    success_rate = (successful / total) * 100

    if success_rate == 100:
        status = "✓ Complete Success"
    elif success_rate >= 80:
        status = "⚠ Partial Success (acceptable)"
    elif success_rate >= 50:
        status = "⚠ Partial Success (concerning)"
    else:
        status = "✗ Majority Failed"

    return {
        'status': status,
        'success_rate': success_rate,
        'successful': successful,
        'faulted': faulted,
        'total': total
    }
```

## Error Handling in Batches (Continued)

### Retry Failed Batches

```python
def retry_failed_batches(orchestrator_url, token, folder_id,
                         release_key, aggregated_results,
                         original_input_batches):
    """Retry batches that failed"""

    failed_batch_ids = []

    # Identify failed batches
    for error_info in aggregated_results['errors']:
        batch_id = error_info.get('batch_id')
        if batch_id is not None:
            failed_batch_ids.append(batch_id)

    if not failed_batch_ids:
        print("No failed batches to retry")
        return []

    print(f"Retrying {len(failed_batch_ids)} failed batches...")

    # Retry each failed batch
    retry_job_ids = []
    for batch_id in failed_batch_ids:
        input_batch = original_input_batches[batch_id]
        jobs = start_process_job(
            orchestrator_url, token, folder_id,
            release_key,
            {'batch_id': batch_id, 'data': input_batch, 'retry': True}
        )
        retry_job_ids.append(jobs[0]['Key'])

    return retry_job_ids
```

```python
def execute_batch_process(orchestrator_url, token, folder_id,
                          release_key, data_to_process, num_batches):
    """
    Complete batch execution workflow

    Args:
        orchestrator_url: Orchestrator URL
        token: Access token
        folder_id: Folder ID
        release_key: Process release key
        data_to_process: List of items to process
        num_batches: Number of parallel jobs

    Returns:
        dict with batch execution results
    """

    # 1. Partition data
    print(f"Partitioning {len(data_to_process)} items into {num_batches} batches...")
    batches = partition_data(data_to_process, num_batches)

    # 2. Start jobs
    print("Starting batch jobs...")
    job_ids, failed_starts = start_batch_jobs_safe(
        orchestrator_url, token, folder_id,
        release_key, batches
    )

    print(f"Started {len(job_ids)} jobs")

    # 3. Monitor jobs
    print("Monitoring job execution...")
    completed_jobs = wait_for_all_jobs_with_progress(
        orchestrator_url, token, folder_id, job_ids
    )
```

## Performance Metrics

### Measuring Batch Performance

```python
def calculate_batch_metrics(start_time, end_time, aggregated_results):
    """Calculate batch execution metrics"""

    total_duration = end_time - start_time
    num_jobs = aggregated_results['total_jobs']
    successful = aggregated_results['successful']

    # Calculate average job duration
    job_durations = []
    for job_id, job in aggregated_results.items():
        if isinstance(job, dict) and 'StartTime' in job:
            duration = calculate_duration(job)
            if duration:
                job_durations.append(duration['seconds'])

    avg_duration = sum(job_durations) / len(job_durations) if job_durations else 0

    return {
        'total_duration': total_duration,
        'avg_job_duration': avg_duration,
        'num_jobs': num_jobs,
        'successful_jobs': successful,
        'throughput': successful / (total_duration / 60),  # jobs/minute
        'parallel_efficiency': avg_duration / total_duration if total_duration > 0 else 0
    }
```

```python
def execute_staged_batch(orchestrator_url, token, folder_id,
                         release_key, batches, jobs_per_wave=5):
    """Start jobs in waves to avoid overwhelming system"""

    all_job_ids = []

    for wave_num in range(0, len(batches), jobs_per_wave):
        wave_batches = batches[wave_num:wave_num + jobs_per_wave]

        print(f"Starting wave {wave_num // jobs_per_wave + 1}...")

        wave_job_ids = start_batch_jobs_safe(
            orchestrator_url, token, folder_id,
            release_key, wave_batches
        )

        all_job_ids.extend(wave_job_ids)

        # Brief pause between waves
        time.sleep(2)

    return all_job_ids
```
