## Update API Endpoint

### HTTP PUT Method

**Endpoint:**
```
PUT /odata/Assets({asset_id})
```

**HTTP Method:** PUT (not POST or PATCH)

**Required:**
- Asset ID in URL path
- Complete asset object in request body
- Authorization header
- Folder context header

## Basic Update Pattern

### The Three-Step Process

1. **GET** the asset to retrieve current state and ID
2. **MODIFY** the Value field
3. **PUT** the modified object back

```python
# Step 1: Get current asset
asset = get_asset(url, token, folder_id, "MyAsset")

# Step 2: Modify value
asset['Value'] = "new value"

# Step 3: Update
update_asset(url, token, folder_id, asset['Id'], asset)
```

## Updating Text Assets

### Simple Text Update

```python
import requests

def update_asset(orchestrator_url, token, folder_id, asset_id, asset_data):
    """Update an asset"""

    response = requests.put(
        f"{orchestrator_url}/odata/Assets({asset_id})",
        headers={
            "Authorization": f"Bearer {token}",
            "X-UIPATH-OrganizationUnitId": str(folder_id),
            "Content-Type": "application/json"
        },
        json=asset_data
    )

    response.raise_for_status()
    return response.json()
```

## Validation Before Update

### Type Validation

```python
def validate_asset_update(asset, new_value):
    """Validate update before sending"""

    asset_type = asset['ValueType']

    if asset_type == 'Text':
        if not isinstance(new_value, str):
            raise TypeError(f"Expected str for Text asset, got {type(new_value)}")

    elif asset_type == 'Bool':
        if not isinstance(new_value, bool):
            raise TypeError(f"Expected bool for Bool asset, got {type(new_value)}")

    elif asset_type == 'Integer':
        if not isinstance(new_value, int):
            raise TypeError(f"Expected int for Integer asset, got {type(new_value)}")

    return True
```

---

## Validation Before Update (Continued)

### Business Logic Validation

```python
def validate_business_rules(asset_name, new_value):
    """Validate business constraints"""

    if asset_name == "MaxRetries":
        if new_value < 0 or new_value > 10:
            raise ValueError("MaxRetries must be between 0 and 10")

    elif asset_name == "TimeoutSeconds":
        if new_value < 1 or new_value > 600:
            raise ValueError("TimeoutSeconds must be between 1 and 600")

    elif asset_name == "EmailServer":
        if not new_value.startswith("smtp."):
            raise ValueError("EmailServer must start with 'smtp.'")

    return True
```

## Safe Update Pattern

### With Validation and Error Handling

```python
def safe_update_asset(orchestrator_url, token, folder_id,
                      asset_name, new_value):
    """Safely update asset with validation"""

    try:
        # Get current asset
        asset = get_asset(orchestrator_url, token, folder_id, asset_name)

        # Validate type
        validate_asset_update(asset, new_value)

        # Validate business rules
        validate_business_rules(asset_name, new_value)

        # Store old value for rollback
        old_value = asset['Value']

        # Update value
        asset['Value'] = new_value

        # Perform update
        updated = update_asset(
            orchestrator_url, token, folder_id,
            asset['Id'], asset
        )

        print(f"✓ Updated {asset_name}: {old_value} → {new_value}")
        return updated

    except Exception as e:
        print(f"✗ Failed to update {asset_name}: {e}")
        raise
```

## Bulk Asset Updates

### Updating Multiple Assets

```python
def bulk_update_assets(orchestrator_url, token, folder_id, updates):
    """
    Update multiple assets

    Args:
        updates: Dict of {asset_name: new_value}
    """

    results = {
        'successful': [],
        'failed': []
    }

    for asset_name, new_value in updates.items():
        try:
            safe_update_asset(
                orchestrator_url, token, folder_id,
                asset_name, new_value
            )
            results['successful'].append(asset_name)
        except Exception as e:
            results['failed'].append({
                'asset': asset_name,
                'error': str(e)
            })

    return results
```

## Conditional Updates

### Update Based on Current Value

```python
def conditional_update(orchestrator_url, token, folder_id,
                       asset_name, condition_fn, new_value):
    """
    Update asset only if condition is met

    Args:
        condition_fn: Function that takes current value and returns bool
    """

    asset = get_asset(orchestrator_url, token, folder_id, asset_name)
    current_value = asset['Value']

    # Check condition
    if condition_fn(current_value):
        asset['Value'] = new_value
        update_asset(orchestrator_url, token, folder_id, asset['Id'], asset)
        print(f"✓ Updated {asset_name}: {current_value} → {new_value}")
        return True
    else:
        print(f"⊘ Skipped {asset_name}: condition not met")
        return False
```

## Atomic Updates with Transactions

### Pattern for Critical Updates

```python
class AssetTransaction:
    """Transaction-like pattern for asset updates"""

    def __init__(self, orchestrator_url, token, folder_id):
        self.orchestrator_url = orchestrator_url
        self.token = token
        self.folder_id = folder_id
        self.rollback_data = []

    def update(self, asset_name, new_value):
        """Add update to transaction"""
        # Get current asset
        asset = get_asset(
            self.orchestrator_url, self.token,
            self.folder_id, asset_name
        )

        # Store for rollback
        self.rollback_data.append({
            'asset_id': asset['Id'],
            'asset_name': asset_name,
            'old_value': asset['Value'],
            'asset_object': dict(asset)
        })

        # Update
        asset['Value'] = new_value
        update_asset(
            self.orchestrator_url, self.token,
            self.folder_id, asset['Id'], asset
        )
```

## Atomic Updates with Transactions (Continued)

```python
    def rollback(self):
        """Rollback all updates"""
        print("Rolling back transaction...")

        for item in reversed(self.rollback_data):
            asset = item['asset_object']
            asset['Value'] = item['old_value']

            update_asset(
                self.orchestrator_url, self.token,
                self.folder_id, item['asset_id'], asset
            )

            print(f"  ↺ Rolled back {item['asset_name']}")

        self.rollback_data.clear()
```

**Usage:**
```python
tx = AssetTransaction(url, token, folder_id)
try:
    tx.update("ServiceURL", "https://new-service.com")
    tx.update("ServicePort", 8080)
    # If any error occurs, rollback
except Exception as e:
    print(f"Error: {e}")
    tx.rollback()
```

## Error Handling (Continued)

### Retry on Transient Failures

```python
def update_with_retry(orchestrator_url, token, folder_id,
                      asset_id, asset_data, max_retries=3):
    """Update with retry on transient failures"""

    for attempt in range(max_retries):
        try:
            return update_asset(
                orchestrator_url, token, folder_id,
                asset_id, asset_data
            )
        except requests.exceptions.ConnectionError as e:
            if attempt < max_retries - 1:
                wait_time = 2 ** attempt  # Exponential backoff
                print(f"Connection error, retrying in {wait_time}s...")
                time.sleep(wait_time)
            else:
                raise
```

## Scope Considerations

### Scope is Immutable

**Important Limitation:**
- Cannot change asset scope via API
- Scope is set at creation time
- To change scope: delete and recreate

```python
# ✗ This won't work
asset['ValueScope'] = 'Global'  # Will be ignored
update_asset(url, token, folder_id, asset['Id'], asset)

# ✓ Must delete and recreate
delete_asset(url, token, folder_id, asset['Id'])
create_asset(url, token, folder_id, name, value, scope='Global')
```

## Concurrency Considerations

### Last Write Wins

**No Built-in Locking:**
- Multiple processes can update same asset
- No conflict detection
- Last write overwrites previous

**Example Problem:**
```
Time    Process A                Process B
T1      GET asset (value=10)
T2                                GET asset (value=10)
T3      value = value + 5 (15)
T4                                value = value + 3 (13)
T5      PUT asset (15)
T6                                PUT asset (13)
Result: value=13 (Process A's update lost!)
```

---

## Concurrency Considerations (Continued)

### Mitigation Strategies

**1. Application-Level Locking:**
```python
from threading import Lock

asset_locks = {}

def update_with_lock(asset_name, new_value):
    if asset_name not in asset_locks:
        asset_locks[asset_name] = Lock()

    with asset_locks[asset_name]:
        # Only one thread can update at a time
        safe_update_asset(url, token, folder_id, asset_name, new_value)
```

**2. Read-Modify-Write Close Together:**
```python
# Minimize time between GET and PUT
asset = get_asset(url, token, folder_id, name)
asset['Value'] = new_value
update_asset(url, token, folder_id, asset['Id'], asset)  # Immediate
```

## Use Case: Automated Credential Rotation

### Complete Pattern

```python
import secrets
import string
from datetime import datetime

def generate_secure_password(length=16):
    """Generate cryptographically secure password"""
    alphabet = string.ascii_letters + string.digits + string.punctuation
    return ''.join(secrets.choice(alphabet) for _ in range(length))

def rotate_credential(orchestrator_url, token, folder_id,
                      credential_name, external_system_update_fn):
    """
    Rotate a credential asset

    Args:
        external_system_update_fn: Function to update password in external system
    """

    print(f"Starting credential rotation for {credential_name}...")

    # Generate new password
    new_password = generate_secure_password()

    # Update external system first
    try:
        external_system_update_fn(new_password)
        print("✓ Updated password in external system")
    except Exception as e:
        print(f"✗ Failed to update external system: {e}")
        raise
```

## Best Practices

### 1. Always Validate

```python
# ✓ Good: Validate before update
validate_asset_update(asset, new_value)
update_asset(url, token, folder_id, asset_id, asset)

# ✗ Bad: No validation
asset['Value'] = new_value  # Could be wrong type
update_asset(url, token, folder_id, asset_id, asset)
```

### 2. Use GET-Modify-PUT Pattern

```python
# ✓ Good: Get full object first
asset = get_asset(url, token, folder_id, name)
asset['Value'] = new_value
update_asset(url, token, folder_id, asset['Id'], asset)

# ✗ Bad: Manually construct object
asset = {'Name': name, 'Value': new_value}  # Missing required fields!
update_asset(url, token, folder_id, asset_id, asset)
```

---

## Best Practices (Continued)

### 3. Handle Errors Gracefully

```python
# ✓ Good: Try-except with specific handling
try:
    update_asset(url, token, folder_id, asset_id, asset)
except requests.exceptions.HTTPError as e:
    if e.response.status_code == 404:
        print("Asset deleted, skipping update")
    else:
        raise

# ✗ Bad: Ignore errors

    update_asset(url, token, folder_id, asset_id, asset)
except:
    pass  # Silent failure
```

### 4. Log Updates

```python

logger.info(f"Updating {asset_name}: {old_value} → {new_value}")
```
